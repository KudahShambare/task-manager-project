1. 📇 profiles

Rule IDBusiness RuleEnforcement
PR‑01Each profile corresponds to exactly one Supabase Auth user (auth.users.id).Foreign key profiles.id REFERENCES auth.users(id) ON DELETE CASCADE.
PR‑02Email must be unique across all profiles.UNIQUE constraint on email column + Supabase Auth enforces uniqueness.
PR‑03Email format must comply with RFC 5322.Application‑level (Supabase Auth validates on signup).
PR‑04Password hashing (bcrypt + salt) managed by Supabase Auth.Not stored in profiles; handled by auth.users.
PR‑05A user may be a member, admin, both, or neither.Determined by existence of rows in members and/or admins tables.
PR‑06Two separate emails = two distinct profiles (no automatic linking).
Example: student email + staff email → separate accounts.		Application logic; Supabase Auth does not merge accounts by default.
PR‑07created_at and updated_at are auto‑managed.DEFAULT NOW() + trigger for updated_at.
PR‑08Deleting a profile also deletes its linked member and/or admin rows.Foreign keys in members and admins with ON DELETE CASCADE.
PR‑09Google Sign‑in is the primary authentication method.Configured in Supabase Auth (external rule).

4. 📁 projects

Rule IDBusiness RuleEnforcement
PJ‑01Only admins can create, edit, or delete projects. Members cannot perform these actions.RLS: FOR ALL checks admin existence.
PJ‑02Project name length: 2–120 characters.CHECK (char_length(name) BETWEEN 2 AND 120).
PJ‑03Project description is required (cannot be NULL or empty).description TEXT NOT NULL.
PJ‑04Project start date is required.start_date DATE NOT NULL.
PJ‑05Project deadline is required.deadline DATE NOT NULL.
PJ‑06Each project has a maximum number of members allowed (max_members INT).max_members INT NOT NULL CHECK (max_members >= 1).
PJ‑07Project status – new ENUM project_status: 'draft', 'published', 'archived'. Default = 'draft'.status project_status NOT NULL DEFAULT 'draft'.
PJ‑08Draft visibility: Only creator (admin who created it) or any admin or a project member with 'editor'/'manager' role can see draft projects. Regular members with viewer role cannot see draft projects.RLS policy: visibility based on role and membership.
PJ‑09Only admins can publish a project (change status from 'draft' to 'published').RLS / application logic.
PJ‑10created_by references the admin who created the project.created_by UUID NOT NULL REFERENCES admins(user_id).
PJ‑11created_at and updated_at auto‑managed.DEFAULT NOW() and trigger.

Rule IDBusiness RuleEnforcement
PM‑01A member can be added to a given project only once.Composite primary key (project_id, member_id).
PM‑02Only admins can add members to projects.RLS: insert allowed only if user is admin.
PM‑03Each membership has a role from ENUM project_role: 'viewer', 'editor', 'manager'.role project_role NOT NULL DEFAULT 'viewer'.

PM‑05editor – can create and edit tasks (but not manage members or publish projects).RLS policies for tasks.
PM‑06manager – can manage members (add/remove) and edit all tasks (same as editor on tasks).RLS: extended permissions.

TS‑01Task title length: 2–160 characters.CHECK (char_length(title) BETWEEN 2 AND 160).

8. 💾 SQL DDL (PostgreSQL with Supabase Auth)

-- Enable pgcrypto for UUID generation
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ENUM types
CREATE TYPE project_status AS ENUM ('draft', 'published', 'archived');
CREATE TYPE task_status AS ENUM ('DRAFT', 'TODO', 'IN_PROGRESS', 'DONE');
CREATE TYPE project_role AS ENUM ('viewer', 'editor', 'manager');

-- Profiles table (links to auth.users)
CREATE TABLE public.profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email TEXT UNIQUE NOT NULL,
    full_name TEXT,
    avatar_url TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Members table
CREATE TABLE public.members (
    user_id UUID PRIMARY KEY REFERENCES public.profiles(id) ON DELETE CASCADE,
    student_id TEXT UNIQUE,
    enrollment_date DATE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Admins table
CREATE TABLE public.admins (
    user_id UUID PRIMARY KEY REFERENCES public.profiles(id) ON DELETE CASCADE,
    employee_id TEXT UNIQUE,
    department TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Projects table
CREATE TABLE public.projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL CHECK (char_length(name) BETWEEN 2 AND 120),
    description TEXT NOT NULL,
    start_date DATE NOT NULL,
    deadline DATE NOT NULL,
    max_members INT NOT NULL CHECK (max_members >= 1),
    status project_status NOT NULL DEFAULT 'draft',
    created_by UUID NOT NULL REFERENCES public.admins(user_id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Project members junction table
CREATE TABLE public.project_members (
    project_id UUID NOT NULL REFERENCES public.projects(id) ON DELETE CASCADE,
    member_id UUID NOT NULL REFERENCES public.members(user_id) ON DELETE CASCADE,
    role project_role NOT NULL DEFAULT 'viewer',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (project_id, member_id)
);

-- Tasks table
CREATE TABLE public.tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES public.projects(id) ON DELETE CASCADE,
    title TEXT NOT NULL CHECK (char_length(title) BETWEEN 2 AND 160),
    description TEXT NOT NULL CHECK (char_length(description) BETWEEN 100 AND 1000),
    status task_status NOT NULL DEFAULT 'DRAFT',
    assignee_member_id UUID REFERENCES public.members(user_id) ON DELETE SET NULL,
    created_by_member_id UUID NOT NULL REFERENCES public.members(user_id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Updated_at triggers
CREATE OR REPLACE FUNCTION touch_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_profiles_touch_updated_at BEFORE UPDATE ON public.profiles
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();
CREATE TRIGGER trg_members_touch_updated_at BEFORE UPDATE ON public.members
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();
CREATE TRIGGER trg_admins_touch_updated_at BEFORE UPDATE ON public.admins
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();
CREATE TRIGGER trg_projects_touch_updated_at BEFORE UPDATE ON public.projects
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();
CREATE TRIGGER trg_tasks_touch_updated_at BEFORE UPDATE ON public.tasks
    FOR EACH ROW EXECUTE FUNCTION touch_updated_at();

-- (Optional) RLS policies to be added based on business rules
-- Example: only admins can insert into projects
ALTER TABLE public.projects ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Admins can insert projects" ON public.projects
